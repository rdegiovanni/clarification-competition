\begin{tikzpicture}[node distance=1.15cm and 1.25cm]

% ---------------------------------------------------------------- title
\node[font=\LARGE\bfseries\color{ink}, anchor=west] (title) at (-3.2, 3.55)
  {LAASeR\_Contract \textcolor{muted}{\normalfont\Large\ --- one question that lets the user correct a draft}};

% ---------------------------------------------------------------- task
\node[stage=muted, minimum width=3.6cm, text width=3.1cm, minimum height=3.3cm] (task) at (-1.4,0)
  {\stagetitle{muted}{\faFile*[regular]}{Task}
   Prompt and entry point\\[4pt]
   {\scriptsize\color{muted} may be vague, incomplete or contradictory; hidden tests decide}};

% ---------------------------------------------------------------- question lane
\node[stage=askc, right=1.0cm of task] (audit)
  {\stagetitle{askc}{\faSearch}{Pinpoint audit}
   Lists every defect of the task\\
   {\scriptsize CONTRADICTION · VAGUE · MISSING · ODD, two readings each, impact HIGH or LOW}\\[3pt]
   Shape reference for data structures\\
   Decides {\bfseries ASK} or {\bfseries CODE}};

\node[stage=z3c, right=of audit] (z3)
  {\stagetitle{z3c}{\faCogs}{z3 evidence}
   Stated examples give the signature\\
   The LLM writes SMT-LIB statements; z3 proves a statement and an example \emph{cannot both hold}\\[3pt]
   {\scriptsize or finds an input where two readings of a defect return different values}};

\node[stage=askc, right=of z3, minimum width=6.4cm, text width=5.9cm] (contract)
  {\stagetitle{askc}{\faFileContract}{Contract question}
   Draft written without asking +\\ audit defects + proven conflict\\[3pt]
   {\scriptsize\itshape\color{ink!80}``I will assume f takes a list first and k second, returns the
   integer count of items $>$ k, e.g.\ f([1,5,7],5) = 1. Is that right, or should it\,\dots?''}};

% ---------------------------------------------------------------- user
\node[stage=userc, below=2.3cm of contract, minimum width=6.4cm, text width=5.9cm,
      minimum height=2.7cm] (user)
  {\stagetitle{userc}{\faUser}{Simulated user (one turn)}
   Knows every hidden requirement: one answer corrects the rule, the argument order and the return format, \emph{all at once}};

% ---------------------------------------------------------------- code lane
\node[stage=codec, left=of user, minimum height=2.7cm] (code)
  {\stagetitle{codec}{\faCode}{Code with the answer}
   The answer overrides the text\\
   Checked (python block, entry point, no placeholder), repaired up to 2 times};

\node[stage=codec, left=of code, minimum height=2.7cm] (verify)
  {\stagetitle{codec}{\faVial}{Verify by execution}
   3 implementations run on probe calls; crashes repaired, disputed results arbitrated, fewest problems wins};

\node[stage=codec, left=1.0cm of verify, minimum width=3.6cm, text width=3.1cm,
      minimum height=2.7cm] (fix)
  {\stagetitle{codec}{\faWrench}{Signature}
   collection first\\
   call adapter:\\
   {\scriptsize f(arr, n, k) $\rightarrow$ f(arr, k)}};

% ---------------------------------------------------------------- numbers on stages
\foreach \n/\c/\s in {1/askc/audit, 2/z3c/z3, 3/askc/contract, 4/codec/code, 5/codec/verify, 6/codec/fix}
  \node[num=\c] at ($(\s.north west)+(0.05,-0.05)$) {\n};

% ---------------------------------------------------------------- arrows
\draw[flow=muted] (task) -- (audit);

\draw[flow=userc] (contract) -- node[right, tag=userc, align=left] {\faQuestion\ 1 question} (user);
\draw[flow=userc] (user) -- node[above, tag=userc] {answer} (code);
\draw[flow=codec] (code) -- (verify);
